// ═══════════════════════════════════════════════════════════════════
// ⚠️ لا تنشرها قبل مقارنتها بقواعدك الحالية (Firebase Console ← Firestore ← Rules).
//    قواعدك الحالية فيها مجموعات اللوحة (promo_allow وغيرها) — لا تحذفها.
//    هذه القواعد تخص فقط: trials و new_orders و user_locations.
//
// الهدف: التطبيق (بدون تسجيل دخول) لا يستطيع:
//   - أن يجعل نفسه VIP (vip_user)
//   - أن يعيد تاريخ بداية التجربة (trial_start) أو يمسح trial_used
//   - أن يقرأ أو يعدّل بيانات جهاز آخر غير متوقعة
// واللوحة (المدير المسجَّل دخوله) تستطيع كل شيء.
//
// غيّر ADMIN_EMAIL إلى بريد حساب الدخول للوحة التحكم.
// سكربت المزامنة (GitHub Actions) يستخدم Admin SDK فلا تؤثر عليه هذه القواعد.
// ═══════════════════════════════════════════════════════════════════

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null && request.auth.token.email == "ADMIN_EMAIL";
    }

    // ── سجل التجربة/VIP لكل جهاز (المعرّف = ANDROID_ID) ──
    match /trials/{deviceId} {
      // التطبيق يقرأ سجل جهازه، ويبحث بـ app_set_id بعد إعادة ضبط المصنع
      allow get: if true;
      allow list: if isAdmin() || request.query.limit <= 1;

      // إنشاء التجربة أول مرة: بوقت الخادم فقط، وبدون VIP
      allow create: if isAdmin() || (
            request.resource.data.trial_used == true
         && request.resource.data.trial_start == request.time
         && !('vip_user' in request.resource.data)
         && !('vip_expires' in request.resource.data)
         && !('purchase_revoked' in request.resource.data)
      );

      // التعديل من التطبيق: فقط آخر ظهور + طلب الكود الترويجي (بحالة pending)
      allow update: if isAdmin() || (
            request.resource.data.diff(resource.data).affectedKeys().hasOnly([
              'last_seen', 'app_version', 'app_version_code',
              'vip_request_phone', 'vip_request_country_code', 'vip_request_status',
              'vip_request_date', 'vip_request_device_model', 'vip_request_edited'
            ])
         && (!('last_seen' in request.resource.data.diff(resource.data).affectedKeys())
             || request.resource.data.last_seen == request.time)
         && (!('vip_request_status' in request.resource.data.diff(resource.data).affectedKeys())
             || request.resource.data.vip_request_status == 'pending')
      );

      allow delete: if isAdmin();
    }

    // ── الطلبات الفورية (نسخة 2.4.2 وما بعدها) ──
    match /new_orders/{orderId} {
      allow create: if request.resource.data.keys().hasOnly([
                         'orderId', 'productId', 'purchaseToken', 'purchaseTimeMs',
                         'packageName', 'app', 'appVersion', 'deviceId', 'createdAt'])
                    && request.resource.data.purchaseToken is string
                    && request.resource.data.purchaseToken.size() < 1000
                    && request.resource.data.createdAt == request.time;
      allow read, update, delete: if isAdmin();
    }

    // ── نسخة احتياطية لمواقع المستخدم (المعرّف = ANDROID_ID) ──
    match /user_locations/{deviceId} {
      allow get: if true;
      allow list: if isAdmin() || request.query.limit <= 1;
      allow create, update: if isAdmin() || (
            request.resource.data.android_id == deviceId
         && request.resource.data.last_updated == request.time
      );
      allow delete: if isAdmin();
    }

    // ── إعدادات التطبيق (التحديث الإجباري) — يقرأها الجميع، يعدّلها المدير فقط ──
    match /app_config/{docId} {
      allow get: if true;
      allow write: if isAdmin();
    }

    // ── مزامنة وقت الخادم لجهاز ليس له سجل trials باسمه (نسخة 2.4.2) ──
    match /time_sync/{deviceId} {
      allow get: if true;
      allow create, update: if request.resource.data.keys().hasOnly(['t'])
                            && request.resource.data.t == request.time;
      allow delete: if isAdmin();
    }

    // ... أبقِ هنا بقية قواعدك الحالية لمجموعات اللوحة كما هي ...
  }
}
